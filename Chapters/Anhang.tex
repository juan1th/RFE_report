\subsection{Calculations for the active band pass filter}
Calculation of the transfer function of one filter stage:
\begin{gather}
    U_{R2} = -U_{out}\\
    U_{R3} = U_{C2}\\
    I_{R2}=\frac{U_{R2}}{R_2} \\
    U_{C2} = I_{R2} \cdot Z_2 = \frac{U_{R2}}{sR_2C_2}\\
   U_{C1} = U_{C2} + U_{R2}=  U_{R2}+\frac{ U_{R2}}{sR_2C_2}\\
   I_{C1} =  \frac{U_{C1}}{Z_{C1}}=\Biggl(U_{R2} +\frac{U_{R2}}{sR_2C_2}\Biggr) \cdot sC_1 \\ 
    I_{R3}= \frac{U_{R3}}{R_3} = \frac{U_{R2}}{sR_2C_2R_3}\\
    I_{R1}= I_{C1} + I_{C2} + I_{R3}=sC_1U_{R2}+\frac{sC_1U_{R2}}{R_2C_2}+\frac{U_{R2}}{R_2}+\frac{U_{R2}}{sR_2C_2R_3}\\
    C_1 = C_2 = C\\
    U_{in}= U_{R3} + U_{R1}= R_1 \cdot I_{R1} + \frac{U_{R2}}{sR_2C} =\\
    U_{R2}\Biggl(\frac{1}{sR_2C} + \frac{2R_1}{R_2} + sCR_1 + \frac{R_1}{sR_2R_3C}\Biggr) =\\
    -U_{out}\frac{R3+2sR_1R_3C+s^2R_1R_2R_3C^2+R_1}{sR_2R_3C}\\
    \begin{split}
     H(s)=\frac{U_{out}}{U_{in}}=-\frac{\frac{sR_2R_3C}{R_1+R_3}}{1+\frac{2R_1R_3}{R_1+R_3}sC+\frac{R_1R_2R_3}{R_1+R_3}s^2C^2}\\
     H(P)=-\frac{\frac{R_2R_3C}{R_1+R_3}\omega_0P}{1+\frac{2R_1R_3}{R_1+R_3}\omega_0PC+\frac{R_1R_2R_3}{R_1+R_3}\omega_0^2P^2C^2}  
    \end{split}
\end{gather}
Calculation the get the factor $\alpha$:\\
The denominator of the function \ref{eq:6} is the following.
\begin{gather}
    D(s)=[1+\frac{\alpha s}{Q_i}+(\alpha s)^2] \cdot [1+\frac{1}{Q_i}\Biggl(\frac{s}{\alpha}\Biggr)+\Biggl(\frac{s}{\alpha}\Biggr)^2]\\
    =1+s\Biggl(\frac{\alpha}{Q_i} +\frac{1}{Q_i\alpha}\Biggr) + s^2\Biggl(\alpha^2+\frac{1}{Q_i^2} + \frac{1}{\alpha^2}\Biggr) + s^3 \Biggl(\frac{\alpha}{Q_i}+\frac{1}{Q_i\alpha}\Biggr) + s^4
\end{gather}
\newpage
Coefficient comparison:
\begin{gather}
    s^0: 1=1\\
    s^1: \frac{\alpha}{Q_i} +\frac{1}{Q_i\alpha} = \frac{\alpha}{b}\Delta\Omega\\
    s^2: \alpha^2+\frac{1}{Q_i^2} + \frac{1}{\alpha^2} = 2+ \frac{\Delta\Omega^2}{b}\\
    s^3: \frac{\alpha}{Q_i} +\frac{1}{Q_i\alpha} = \frac{\alpha}{b}\Delta\Omega\\
    s^4: 1=1
\end{gather}
The equation~\ref{eq:16} can be derived from $s^1$ or $s^3$ and must be substituted in the coefficient equation of $s^2$ as follows.
\begin{gather}
    \alpha^2+\Biggl[\frac{\alpha\Delta\Omega a}{b(1+\alpha^2)}\Biggr] + \frac{1}{\alpha^2}-2-\frac{(\Delta\Omega)^2}{b}=0\label{eq:89}
\end{gather}
The equation~\ref{eq:89} was solved using Wolfram Alpha \ref{fig:96}\\
Further necessary values for the calculations are:\\
\begin{gather}
    Q=\frac{f_m}{f_u-f_l}=4.47\\
    \Delta\Omega=\frac{1}{Q}\\
    a = \sqrt{2}; 
    b=1 \\ \text{a and b are given for a Butterworth filter}
\end{gather}

\begin{lstlisting}[style=mystyle, language=Python, label={lst:load-code}]
import cmath
import math

Z0 = 50  # characteristic impedance in ohms

mag = float(input("Enter reflection coefficient magnitude |Gamma|: "))
phase_deg = float(input("Enter reflection coefficient phase in degrees: "))

phase_rad = math.radians(phase_deg)

Gamma = mag * cmath.exp(1j * phase_rad)

ZL = Z0 * (1 + Gamma) / (1 - Gamma)

print(f"\nGamma = {Gamma.real:.6f} + j{Gamma.imag:.6f}")
print(f"Load impedance ZL = {ZL.real:.6f} + j{ZL.imag:.6f} ohms")
\end{lstlisting}


\begin{lstlisting}[style=mystyle, language=Python, label={lst:smith-code}]
	import matplotlib.pyplot as plt
	import pysmithchart
	
	Z0 = 50
	
	Z = [
	1e9 + 0j,   # Open
	0 + 0j,     # Short
	100 + 0j,   # 100 ohm
	25 + 0j,    # 25 ohm
	50 + 0j,    # 50 ohm
	0 - 50j     # Capacitive load
	]
	
	labels = ["Open", "Short", "100Ω", "25Ω", "50Ω", "Capacitive"]
	colors = ["red", "blue", "green", "orange", "purple", "brown"]
	
	plt.figure(figsize=(9, 9))
	ax = plt.subplot(111, projection="smith", Z0=Z0)
	
	for z, label, color in zip(Z, labels, colors):
	ax.plot([z], marker="o", markersize=10, color=color, ls="")
	ax.text(z, label, color=color)
	
	ax.set_title("Smith Chart of 6 Load Impedances")
	plt.tight_layout()
	plt.show()
\end{lstlisting}
